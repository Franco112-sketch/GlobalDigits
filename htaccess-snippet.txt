# Apache (.htaccess) — add this to the .htaccess file in your site's
# public/root folder, alongside index.html / landing.html.
# These are internal rewrites (no [R] flag), so the browser's address
# bar keeps showing "/" or "/app" — it never sees the real filename.

RewriteEngine On

# Root "/" quietly serves landing.html's content
RewriteRule ^$ landing.html [L]

# "/app" quietly serves index.html's content (the app)
RewriteRule ^app/?$ index.html [L]

# Prevent the real filenames from being reachable directly, so visiting
# "/index.html" or "/landing.html" by URL still works for you to test,
# but you can 301-redirect them to the clean paths once you've verified
# the rewrites above are working, e.g.:
# RewriteRule ^index\.html$ /app [R=301,L]
# RewriteRule ^landing\.html$ / [R=301,L]
